window.startMediaBunnyExport = async function(exportW, exportH, fps, startTime, endTime, format = 'webm') {
            mbExportFormat = format;
            const timeStep = 1 / fps;
            const totalFrames = Math.ceil((endTime - startTime) / timeStep);
            let currentFrame = 0;
            
            State.currentTime = startTime;
            
            // Use OffscreenCanvas for main export canvas
            const exportCanvas = createOffscreenCanvas(exportW, exportH);
            const exportCtx = exportCanvas.getContext('2d');
            
            // --- Audio setup: Pre-render audio using OfflineAudioContext ---
            let audioCaptureBuffer = null;
            const audioSampleRate = 48000;
            const audioChunkDuration = 0.1; // 100ms per audio chunk
            
            // Save speaker gain before muting
            const prevSpeakerGain = State.speakerNode ? State.speakerNode.gain.value : 1;
            
            try {
                // Find audio clips that overlap with the export range and have decoded buffers
                const audioClips = State.clips.filter(c => {
                    if (c.hidden) return false;
                    if (!c.buffer) return false;
                    const track = State.tracks.find(t => t.id === c.trackId);
                    if (track && track.hidden) return false;
                    const clipStart = Math.max(c.start, startTime);
                    const clipEnd = Math.min(c.start + c.duration, endTime);
                    return clipEnd > clipStart;
                });
                
                if (audioClips.length > 0) {
                    const totalDuration = endTime - startTime;
                    const totalSamples = Math.ceil(totalDuration * audioSampleRate);
                    const offlineCtx = new OfflineAudioContext(2, totalSamples, audioSampleRate);
                    
                    const masterGain = offlineCtx.createGain();
                    masterGain.gain.value = 1;
                    masterGain.connect(offlineCtx.destination);
                    
                    audioClips.forEach(c => {
                        const track = State.tracks.find(t => t.id === c.trackId);
                        const clipStart = Math.max(c.start, startTime);
                        const clipEnd = Math.min(c.start + c.duration, endTime);
                        const sourceOffset = c.sourceOffset + (clipStart - c.start);
                        const duration = clipEnd - clipStart;
                        
                        const source = offlineCtx.createBufferSource();
                        source.buffer = c.buffer;
                        
                        const gainNode = offlineCtx.createGain();
                        const vol = (track && track.volume !== undefined) ? track.volume : (c.volume || 1);
                        gainNode.gain.value = vol;
                        
                        source.connect(gainNode);
                        gainNode.connect(masterGain);
                        
                        const startTimeInOffline = clipStart - startTime;
                        source.start(startTimeInOffline, sourceOffset, duration);
                    });
                    
                    console.log('Rendering audio offline (' + totalDuration.toFixed(1) + 's)...');
                    const renderedBuffer = await offlineCtx.startRendering();
                    audioCaptureBuffer = renderedBuffer;
                    console.log('Audio pre-rendered OK:', audioCaptureBuffer.duration.toFixed(2), 'seconds');
                }
            } catch (e) {
                console.warn('Audio pre-render failed, continuing without audio:', e);
            }
            
            // Mute speakers during export (but keep audio flowing if capturing)
            if (State.speakerNode) State.speakerNode.gain.value = 0;
            
            // Helper to recursively collect active video clips
            const getActiveVideoClips = (sceneId, time) => {
                let list = [];
                State.clips.forEach(c => {
                    const track = State.tracks.find(tr => tr.id === c.trackId);
                    if (c.hidden || (track && track.hidden)) return;
                    
                    const matchesScene = (sceneId === null) ? (!c.sceneId) : (c.sceneId === sceneId);
                    if (!matchesScene) return;
                    
                    if (time >= c.start && time < c.start + c.duration) {
                        if (c.type === 'video') {
                            list.push({ clip: c, timeInClip: time - c.start });
                        } else if (c.type === 'scene') {
                            // Recursively find video clips inside the nested scene
                            const subSceneClips = getActiveVideoClips(c.id, time - c.start);
                            list = list.concat(subSceneClips);
                        }
                    }
                });
                return list;
            };

            const recreateVideoEl = async (clip) => {
                if (!clip.fileUrl) return null;
                const oldVideo = clip.videoEl;
                const wasMuted = oldVideo ? oldVideo.muted : true;
                const volume = oldVideo ? oldVideo.volume : 1;
                
                const newVideo = document.createElement('video');
                newVideo.src = clip.fileUrl;
                newVideo.crossOrigin = "anonymous";
                newVideo.muted = wasMuted;
                newVideo.volume = volume;
                
                const loadPromise = new Promise((resolve) => {
                    newVideo.onloadedmetadata = () => {
                        // Ensure we have enough data to seek
                        if (newVideo.readyState >= 2) resolve();
                        else {
                            newVideo.addEventListener('canplay', () => resolve(), { once: true });
                        }
                    };
                    newVideo.onerror = () => resolve();
                });
                newVideo.load();
                await loadPromise;
                
                clip.videoEl = newVideo;
                return newVideo;
            };

            // Improved helper to wait for all active video clips to seek
            const waitForSeeking = async (t) => {
                const activeVideos = getActiveVideoClips(State.activeSceneId, t);
                if (activeVideos.length === 0) return;

                const seekPromises = activeVideos.map(async ({ clip, timeInClip }) => {
                    let el = clip.videoEl;
                    if (!el) return;

                    // Check for error state
                    if (el.error) {
                        console.warn("Video element in error state before seek, recreating...", clip.title);
                        el = await recreateVideoEl(clip);
                        if (!el) return;
                    }
                    
                    const targetTime = clip.sourceOffset + timeInClip;
                    
                    // Skip if already at target
                    if (Math.abs(el.currentTime - targetTime) < 0.01 && el.readyState >= 2) {
                        return;
                    }
                    
                    // Ensure video element has loaded enough data before seeking
                    if (el.readyState < 2) {
                        const ready = await ensureVideoReady(el);
                        if (!ready) {
                            console.warn("Could not prepare video for seek, recreating...", clip.title);
                            el = await recreateVideoEl(clip);
                            if (!el) return;
                        }
                    }
                    
                    const performSeek = (videoEl) => {
                        return new Promise(resolve => {
                            let resolved = false;
                            const seekTimeout = 5000;
                            const timeoutId = setTimeout(() => {
                                if (resolved) return;
                                resolved = true;
                                videoEl.removeEventListener('seeked', onSeeked);
                                videoEl.removeEventListener('error', onError);
                                console.warn("Video seek timeout, forcing resolve:", targetTime);
                                resolve(false);
                            }, seekTimeout);

                            const onSeeked = () => {
                                if (resolved) return;
                                resolved = true;
                                clearTimeout(timeoutId);
                                videoEl.removeEventListener('seeked', onSeeked);
                                videoEl.removeEventListener('error', onError);
                                resolve(true);
                            };
                            const onError = () => {
                                if (resolved) return;
                                resolved = true;
                                clearTimeout(timeoutId);
                                videoEl.removeEventListener('seeked', onSeeked);
                                videoEl.removeEventListener('error', onError);
                                console.warn("Video seek error during performSeek (target:", targetTime, ")");
                                resolve(false);
                            };
                            videoEl.addEventListener('seeked', onSeeked);
                            videoEl.addEventListener('error', onError);
                            videoEl.currentTime = targetTime;
                        });
                    };

                    let success = await performSeek(el);
                    if (!success) {
                        console.warn("Seeking failed or timed out. Recreating video element and retrying...", clip.title);
                        el = await recreateVideoEl(clip);
                        if (el) {
                            await performSeek(el);
                        }
                    }
                });

                await Promise.all(seekPromises);
            };

            // Pre-seek video elements to the start time before the frame loop.
            // This ensures video frames are decoded and ready so the first 3-4
            // exported frames don't come out empty while video elements load.
            await waitForSeeking(startTime);

            // Configure worker
            worker.postMessage({
                type: 'start',
                config: {
                    width: exportW,
                    height: exportH,
                    fps: fps,
                    bitrate: format === 'mp4' ? 8e6 : 5e6,
                    format: format,
                    hasAudio: audioCaptureBuffer !== null
                }
            });
            
            // Wait for worker to be ready
            await new Promise((resolve, reject) => {
                const readyHandler = (e) => {
                    if (e.data.type === 'ready') {
                        worker.removeEventListener('message', readyHandler);
                        resolve();
                    } else if (e.data.type === 'error') {
                        worker.removeEventListener('message', readyHandler);
                        reject(new Error(e.data.error));
                    }
                };
                worker.addEventListener('message', readyHandler);
                setTimeout(() => {
                    worker.removeEventListener('message', readyHandler);
                    resolve(); // Continue even if ready not received
                }, 5000);
            });
            console.log('Worker ready, starting export...');
            
            // Global error listener for worker during export
            let workerError = null;
            const workerMsgHandler = (e) => {
                if (e.data.type === 'error') {
                    workerError = e.data.error;
                }
            };
            worker.addEventListener('message', workerMsgHandler);
            
            // Audio chunk tracking
            let audioChunkIndex = 0;
            let audioSentUpTo = 0;
            
            for (let t = startTime; t < endTime; t += timeStep) {
                if (State.cancelExport || workerError) {
                    worker.postMessage({ type: 'cancel' });
                    break;
                }
                
                State.currentTime = t;
                
                // Wait for video seeking to complete
                await waitForSeeking(t);
                
                // Draw canvas at this timestamp
                drawCanvas(exportCtx, exportW, exportH);
                
                // Extract bitmap
                const bitmap = await createImageBitmap(exportCanvas);
                
                // Send audio chunks up to current video position
                if (audioCaptureBuffer) {
                    const currentAudioTime = t - startTime;
                    while (audioSentUpTo < currentAudioTime && audioSentUpTo < audioCaptureBuffer.duration) {
                        const startSample = Math.floor(audioSentUpTo * audioSampleRate);
                        const frameCount = Math.min(
                            Math.floor(audioChunkDuration * audioSampleRate),
                            audioCaptureBuffer.length - startSample
                        );
                        if (frameCount <= 0) break;
                        
                        const left = audioCaptureBuffer.getChannelData(0);
                        const right = audioCaptureBuffer.getChannelData(1);
                        const interleaved = new Float32Array(frameCount * 2);
                        for (let i = 0; i < frameCount; i++) {
                            interleaved[i * 2] = left[startSample + i];
                            interleaved[i * 2 + 1] = right[startSample + i];
                        }
                        
                        try {
                            const audioData = new AudioData({
                                format: 'f32',
                                sampleRate: audioSampleRate,
                                numberOfChannels: 2,
                                numberOfFrames: frameCount,
                                timestamp: audioSentUpTo * 1e6,
                                data: interleaved
                            });
                            
                            worker.postMessage({
                                type: 'audio-data',
                                audioData: audioData,
                                timestamp: audioSentUpTo,
                                index: audioChunkIndex++
                            }, [audioData]);
                            
                            audioSentUpTo += frameCount / audioSampleRate;
                        } catch (audioErr) {
                            console.warn('Audio chunk send failed, stopping audio:', audioErr);
                            audioCaptureBuffer = null;
                            break;
                        }
                    }
                }
                
                // Wait for worker acknowledgement before generating next frame
                const frameProcessedPromise = new Promise((resolve, reject) => {
                    const handleMessage = (e) => {
                        if (e.data.type === 'frame-processed' && e.data.index === currentFrame) {
                            worker.removeEventListener('message', handleMessage);
                            resolve();
                        } else if (e.data.type === 'error') {
                            worker.removeEventListener('message', handleMessage);
                            reject(new Error(e.data.error));
                        }
                    };
                    worker.addEventListener('message', handleMessage);
                });

                // Send frame to worker
                worker.postMessage({
                    type: 'frame',
                    bitmap: bitmap,
                    timestamp: (t - startTime),
                    duration: timeStep,
                    index: currentFrame
                }, [bitmap]);
                
                try {
                    await frameProcessedPromise;
                } catch (err) {
                    workerError = err.message;
                    worker.postMessage({ type: 'cancel' });
                    break;
                }

                currentFrame++;
                
                // Update progress bar
                const progress = Math.min(100, Math.round((currentFrame / totalFrames) * 100));
                document.getElementById('exportProgressBar').style.width = `${progress}%`;
                document.getElementById('exportProgressText').textContent = `${progress}%`;
                
                // Yield thread to keep UI smooth and check cancellation
                await new Promise(resolve => setTimeout(resolve, 0));
            }
            
            // Helper to flush remaining audio chunks
            const flushAudioChunks = () => {
                if (audioCaptureBuffer && audioSentUpTo < audioCaptureBuffer.duration) {
                    while (audioSentUpTo < audioCaptureBuffer.duration) {
                        const startSample = Math.floor(audioSentUpTo * audioSampleRate);
                        const frameCount = Math.min(
                            Math.floor(audioChunkDuration * audioSampleRate),
                            audioCaptureBuffer.length - startSample
                        );
                        if (frameCount <= 0) break;
                        
                        const left = audioCaptureBuffer.getChannelData(0);
                        const right = audioCaptureBuffer.getChannelData(1);
                        const interleaved = new Float32Array(frameCount * 2);
                        for (let i = 0; i < frameCount; i++) {
                            interleaved[i * 2] = left[startSample + i];
                            interleaved[i * 2 + 1] = right[startSample + i];
                        }
                        
                        try {
                            const audioData = new AudioData({
                                format: 'f32',
                                sampleRate: audioSampleRate,
                                numberOfChannels: 2,
                                numberOfFrames: frameCount,
                                timestamp: audioSentUpTo * 1e6,
                                data: interleaved
                            });
                            
                            worker.postMessage({
                                type: 'audio-data',
                                audioData: audioData,
                                timestamp: audioSentUpTo,
                                index: audioChunkIndex++
                            }, [audioData]);
                            
                            audioSentUpTo += frameCount / audioSampleRate;
                        } catch (audioErr) {
                            console.warn('Audio flush failed, stopping audio:', audioErr);
                            break;
                        }
                    }
                }
            };
            
            const restoreSpeaker = () => {
                if (State.speakerNode) {
                    State.speakerNode.gain.value = prevSpeakerGain;
                }
            };
            
            const cleanupExport = () => {
                worker.removeEventListener('message', workerMsgHandler);
                State.isExporting = false;
                restoreSpeaker();
                closeExportModal();
                stopMedia();
            };

            if (workerError) {
                flushAudioChunks();
                alert(`Export failed: ${workerError}`);
                cleanupExport();
            } else if (State.cancelExport) {
                flushAudioChunks();
                cleanupExport();
            } else {
                // Flush remaining audio chunks
                flushAudioChunks();
                
                // Post finalize message to trigger worker to encode remaining frames and build output
                worker.postMessage({ type: 'finalize' });
                
                // Wait for done/error with timeout
                try {
                    const doneResult = await new Promise((resolve, reject) => {
                        const finalizeDoneHandler = (e) => {
                            if (e.data.type === 'done') {
                                worker.removeEventListener('message', finalizeDoneHandler);
                                worker.removeEventListener('message', finalizeErrorHandler);
                                resolve(e.data);
                            }
                        };
                        const finalizeErrorHandler = (e) => {
                            if (e.data.type === 'error') {
                                worker.removeEventListener('message', finalizeDoneHandler);
                                worker.removeEventListener('message', finalizeErrorHandler);
                                reject(new Error(e.data.error));
                            }
                        };
                        worker.addEventListener('message', finalizeDoneHandler);
                        worker.addEventListener('message', finalizeErrorHandler);
                        
                        // Timeout after 2 minutes
                        const timeoutId = setTimeout(() => {
                            worker.removeEventListener('message', finalizeDoneHandler);
                            worker.removeEventListener('message', finalizeErrorHandler);
                            console.error('Finalize timed out after 120 seconds');
                            reject(new Error('Finalize timed out after 120 seconds'));
                        }, 120000);
                    });
                    
                    // Done received - download blob
                    const mimeType = format === 'mp4' ? 'video/mp4' : 'video/webm';
                    const ext = format === 'mp4' ? 'mp4' : 'webm';
                    const blob = new Blob([doneResult.buffer], { type: mimeType });
                    const url = URL.createObjectURL(blob);
                    const a = document.createElement('a');
                    a.href = url;
                    a.download = `StudioPro_Export_MediaBunny.${ext}`;
                    a.click();
                    URL.revokeObjectURL(url);
                    
                    cleanupExport();
                } catch (err) {
                    alert(`Export failed: ${err.message}`);
                    cleanupExport();
                }
            }
        }
